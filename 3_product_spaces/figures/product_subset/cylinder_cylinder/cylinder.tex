\tikzmath{
  function x(\x, \y, \z, \theta, \phi) {
    \c = cos(\theta);
    \s = sin(\theta);
    \xa = +\c * \x + \s * \z;
    \ya = \y;
    \za = -\s * \x + \c * \z;
    %
    \xb = \xa;
    %\yb = +cos(\phi) * \ya + sin(\phi) * \za;
    %\zb = -sin(\phi) * \ya + cos(\phi) * \za;
    %  
    return \xb;
  };
  function y(\x, \y, \z, \theta, \phi) {
    \c = cos(\theta);
    \s = sin(\theta);
    \xa = +\c * \x + \s * \z;
    \ya = \y;
    \za = -\s * \x + \c * \z;
    %
    %\xb = \xa;
    \yb = +cos(\phi) * \ya + sin(\phi) * \za;
    %\zb = -sin(\phi) * \ya + cos(\phi) * \za;
    %   
    return \yb;
  };
}
