\tikzmath{
  function joint(\x, \y, \N, \c) {
    \a=3;
    \b=3;
    return exp( (\y + \a - 1) * ln(\x) + (\N - \y + \b - 1) * ln(1 - \x) + 3.401197 + \c);
  };
}
