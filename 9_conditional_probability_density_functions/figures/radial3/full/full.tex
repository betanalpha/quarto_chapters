\tikzmath{
  function target(\x, \y) {
\a = 4.00000000;
\b = 5.00000000;
\C = 4.64599218;
\k = 2.00000000;
\D = 0.06981750;
    \r = sqrt(\x * \x + \y * \y);
    \theta = atan2(\y, \x);
    return \D * exp( \C + (\a - 1) * ln(\r) - \b * \r + \k * cos( \theta - 60 )  );
  };
}
